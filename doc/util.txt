=== Documentação do GSA (util.h) ===

[Função]: attPesos(const vector<vector<float>>& pesos, const vector<vector<float>>& grad, float taxa, float lambda = 1e-3f);
[Retorno]: vector<vector<float>>
[Fórmula]: nova[i][j] = pesos[i][j] - taxa * grad[i][j] - lambda * pesos[i][j]

[Função]: attPesosMomentum(const vector<vector<float>>& pesos, const vector<vector<float>>& grad, float taxa, float momento, vector<vector<float>>& velocidade);
[Retorno]: vector<vector<float>>
[Fórmula]: velocidade[i][j] = momento * velocidade[i][j] + grad[i][j]
           nova[i][j] = pesos[i][j] - taxa * velocidade[i][j]

[Função]: attPesosAdam(const vector<vector<float>>& pesos, const vector<vector<float>>& grad, vector<vector<float>>& m, vector<vector<float>>& v, float taxa, float beta1 = 0.9f, float beta2 = 0.999f, float eps = 1e-8f, int iteracao = 1, float lambda = 0.001f);
[Retorno]: vector<vector<float>>
[Fórmula]: m[i][j] = beta1 * m[i][j] + (1 - beta1) * g
           v[i][j] = beta2 * v[i][j] + (1 - beta2) * g²
           mChapeu = m[i][j] / (1 - beta1^iteracao)
           vChapeu = v[i][j] / (1 - beta2^iteracao)
           nova[i][j] = pesos[i][j] - taxa * mChapeu / (√vChapeu + eps)

[Função]: attPesosAdam1D(vector<float>& p, const vector<float>& grad, vector<float>& m, vector<float>& v, float taxa, float beta1 = 0.9f, float beta2 = 0.999f, float eps = 1e-8f, int t = 1, float lambda = 0.001f);
[Retorno]: vector<float>
[Fórmula]: Versão 1D do Adam (mesma fórmula acima)

[Função]: regularL1(const vector<vector<float>>& pesos, float lambda);
[Retorno]: vector<vector<float>>
[Fórmula]: res[i][j] = lambda * sign(pesos[i][j])

[Função]: regularL2(const vector<vector<float>>& pesos, float lambda);
[Retorno]: vector<vector<float>>
[Fórmula]: res[i][j] = lambda * pesos[i][j]

[Função]: dropout(vector<vector<float>> tensor, float taxa);
[Retorno]: vector<vector<float>>
[Fórmula]: valor = 0 (com probabilidade taxa) ou valor/(1-taxa)

[Função]: normEntrada(const vector<float>& vetor);
[Retorno]: vector<float>
[Fórmula]: res[i] = (vetor[i] - min) / (max - min)

[Função]: normZPonto(const vector<float>& v);
[Retorno]: vector<float>
[Fórmula]: res[i] = (v[i] - média) / desvio_padrão

[Função]: acuracia(const vector<vector<float>>& saida, const vector<vector<float>>& esperado);
[Retorno]: float
[Fórmula]: corretos/total

[Função]: precisao(const vector<vector<int>>& confusao);
[Retorno]: float
[Fórmula]: TP / (TP + FP)

[Função]: recall(const vector<vector<int>>& confusao);
[Retorno]: float
[Fórmula]: TP / (TP + FN)

[Função]: f1Ponto(const vector<vector<int>>& confusao);
[Retorno]: float
[Fórmula]: 2 * (precisão * recall) / (precisão + recall)

[Função]: mse(const vector<float>& saida, const vector<float>& esperado);
[Retorno]: float
[Fórmula]: Σ(saida[i] - esperado[i])² / n

[Função]: derivadaMse(const vector<float>& saida, const vector<float>& esperado);
[Retorno]: vector<float>
[Fórmula]: derivada de (y-ŷ)² = 2*(y-ŷ)

[Função]: klDivergencia(const vector<float>& p, const vector<float>& q);
[Retorno]: float
[Fórmula]: Σ p[i] * log(p[i] / q[i])

[Função]: rocAuc(const vector<float>& pontos, const vector<int>& rotulos);
[Retorno]: float
[Fórmula]: Área sob a curva ROC

[Função]: erroAbsolutoMedio(const vector<float>& saida, const vector<float>& esperado);
[Retorno]: float
[Fórmula]: Σ |saida[i] - esperado[i]| / n

[Função]: erroQuadradoEsperado(const vector<float>& saida, const vector<float>& esperado);
[Retorno]: float
[Fórmula]: 0.5 * Σ (saida[i] - esperado[i])²

[Função]: derivadaErro(const vector<float>& saida, const vector<float>& esperado);
[Retorno]: vector<float>
[Fórmula]: deriv[i] = saida[i] - esperado[i]

[Função]: entropiaCruzada(const vector<float>& y, const vector<float>& yChapeu);
[Retorno]: float
[Fórmula]: -Σ y[i] * log(yChapeu[i])

[Função]: derivadaEntropiaCruzada(const vector<float>& y, const vector<float>& yChapeu);
[Retorno]: vector<float>
[Fórmula]: deriv[i] = yChapeu[i] - y[i]

[Função]: huberPerda(const vector<float>& saida, const vector<float>& esperado, float delta = 1.0f);
[Retorno]: float
[Fórmula]: Lδ(a) = 0.5a² se |a|≤δ, δ(|a|-0.5δ) caso contrário

[Função]: derivadaHuber(const vector<float>& saida, const vector<float>& esperado, float delta = 1.0f);
[Retorno]: vector<float>
[Fórmula]: deriv[i] = diff se |diff|≤δ, δ*sign(diff) caso contrário

[Função]: perdaTripleto(const vector<float>& ancora, const vector<float>& positiva, const vector<float>& negativa, float margem = 1.0f);
[Retorno]: float
[Fórmula]: max(0, dist(ancora, positiva) - dist(ancora, negativa) + margem)

[Função]: contrastivaPerda(const vector<float>& saida1, const vector<float>& saida2, int rotulo, float margem = 1.0f);
[Retorno]: float
[Fórmula]: dist se rotulo=1, max(0, margem - √dist) se rotulo=0

[Função]: softmax(const vector<float>& arr, float temp = 1.0f);
[Retorno]: vector<float>
[Fórmula]: exp(arr[i]/temp) / Σ exp(arr[j]/temp)

[Função]: derivadaSoftmax(const vector<float>& arr, const vector<float>& gradSaida);
[Retorno]: vector<float>
[Fórmula]: arr[i] * (gradSaida[i] - Σ arr[j]*gradSaida[j])

[Função]: softmaxLote(const vector<vector<float>>& m, float temp = 1.0f);
