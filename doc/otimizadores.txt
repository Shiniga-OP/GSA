=== Documentação do GSA (otimizadores.h) ===

[Classe]: Otimizador {
[Construtor]: abstrata/nula
[Função]: att(vector<vector<float>>& pesos,vector<vector<float>>& gradPesos, vector<float>& bias, vector<float>& gradBias)
[Retorno]: void
[Fórmula]: nenhuma
}

[Classe]: Adam -> Otimizador {
[Construtor]: float taxa = 0.001f, float b1 = 0.9f, float b2 = 0.999f, float eps = 1e-8f
[Função]: att(vector<vector<float>>& pesos,vector<vector<float>>& gradPesos, vector<float>& bias, vector<float>& gradBias)
[Retorno]: void
[Fórmula]: t++; m = b1*m + (1-b1)*g; v = b2*v + (1-b2)*g²; m̂ = m/(1-b1^t); v̂ = v/(1-b2^t); θ = θ - α*m̂/(√v̂ + ε)
}

[Classe]: SGD -> Otimizador {
[Construtor]: float taxa = 0.01f, float momentum = 0.0f
[Função]: att(vector<vector<float>>& pesos,vector<vector<float>>& gradPesos, vector<float>& bias, vector<float>& gradBias)
[Retorno]: void
[Fórmula]: v = μ*v - α*g; θ = θ + v
}

[Classe]: AdaGrad -> Otimizador {
[Construtor]: float taxa = 0.01f, float eps = 1e-8f
[Função]: att(vector<vector<float>>& pesos,vector<vector<float>>& gradPesos, vector<float>& bias, vector<float>& gradBias)
[Retorno]: void
[Fórmula]: G += g²; θ = θ - α*g/(√G + ε)
}

[Classe]: RMSprop -> Otimizador {
[Construtor]: float taxa = 0.001f, float decaimento = 0.9f, float eps = 1e-8f
[Função]: att(vector<vector<float>>& pesos,vector<vector<float>>& gradPesos, vector<float>& bias, vector<float>& gradBias)
[Retorno]: void
[Fórmula]: E[g²] = ρ*E[g²] + (1-ρ)*g²; θ = θ - α*g/(√E[g²] + ε)
}

[Classe]: AdaDelta -> Otimizador {
[Construtor]: float rho = 0.95f, float eps = 1e-6f
[Função]: att(vector<vector<float>>& pesos,vector<vector<float>>& gradPesos, vector<float>& bias, vector<float>& gradBias)
[Retorno]: void
[Fórmula]: E[g²] = ρ*E[g²] + (1-ρ)*g²; Δx = √(E[Δx²] + ε)/√(E[g²] + ε)*g; E[Δx²] = ρ*E[Δx²] + (1-ρ)*Δx²; θ = θ - Δx
}

[Classe]: Nesterov -> Otimizador {
[Construtor]: float taxa = 0.01f, float momentum = 0.9f
[Função]: att(vector<vector<float>>& pesos,vector<vector<float>>& gradPesos, vector<float>& bias, vector<float>& gradBias)
[Retorno]: void
[Fórmula]: v_antigo = v; v = μ*v - α*g; θ = θ - μ*v_antigo + (1+μ)*v
}

[Classe]: AdamW -> Adam {
[Construtor]: float taxa = 0.001f, float pesoDecaimento = 0.01f
[Função]: att(vector<vector<float>>& pesos,vector<vector<float>>& gradPesos, vector<float>& bias, vector<float>& gradBias)
[Retorno]: void
[Fórmula]: θ = θ - α*λ*θ (decay); depois aplica Adam normal
}
